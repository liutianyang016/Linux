#include <sqlite3.h>
#include <stdio.h>
#include "database.h"

sqlite3 *db;

int init_database() {
    int rc = sqlite3_open("data/grades.db", &db);
    
    if(rc) {
        fprintf(stderr, "Can't open database: %s\n", sqlite3_errmsg(db));
        return 1;
    }
    
    // 创建表如果不存在
    const char *sql = "CREATE TABLE IF NOT EXISTS students ("
                     "id INTEGER PRIMARY KEY AUTOINCREMENT,"
                     "name TEXT NOT NULL,"
                     "student_id TEXT UNIQUE NOT NULL,"
                     "course TEXT NOT NULL,"
                     "grade REAL);";
    
    rc = sqlite3_exec(db, sql, 0, 0, 0);
    
    if(rc != SQLITE_OK) {
        fprintf(stderr, "SQL error: %s\n", sqlite3_errmsg(db));
        return 1;
    }
    
    return 0;
}

void close_database() {
    sqlite3_close(db);
}

// 添加学生成绩
int add_grade(const char *name, const char *student_id, const char *course, float grade) {
    char sql[256];
    sprintf(sql, "INSERT INTO students (name, student_id, course, grade) "
                "VALUES ('%s', '%s', '%s', %.2f);", 
                name, student_id, course, grade);
    
    char *err_msg = 0;
    int rc = sqlite3_exec(db, sql, 0, 0, &err_msg);
    
    if(rc != SQLITE_OK) {
        fprintf(stderr, "SQL error: %s\n", err_msg);
        sqlite3_free(err_msg);
        return 1;
    }
    
    return 0;
}
